{{- $v := include "es.certVolume" (dict "root" . "kind" "http") | fromYaml }}

     kubectl port-forward -n {{ .Release.Namespace }} svc/{{ $cluster }} 9200:{{ .Values.service.httpPort }}
     curl --cacert ca.crt -u "elastic:$PASSWORD" 'https://localhost:9200/_cluster/health?pretty'
